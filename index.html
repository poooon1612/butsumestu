// ======================================== // // 📖 第4章 // // JavaScriptでARを動かそう！ // // 🏭 保全ポイント // // この章はARを動かすプログラムです。 // // 触る部分は // // VERSION // // のみ！ // // ======================================== // 🟡 展示会前に動画を更新したら // この数字だけ変更します。 const VERSION = 2; // 第2章の動画を取得 const movie = document.querySelector("#movie"); // 第3章の展示場所を取得 const target = document.querySelector("[mindar-image-target]"); // キャッシュ対策 movie.src = `movie.mp4?v=${VERSION}`; // ======================================== // // 🎬 作品を認識したら // // 動画を再生 // // ======================================== target.addEventListener("targetFound", async () => { try { movie.currentTime = 0; await movie.play(); } catch (error) { console.error(error); } }); // ======================================== // // 👋 作品が見えなくなったら // // 動画を停止 // // ======================================== target.addEventListener("targetLost", () => { movie.pause(); });
